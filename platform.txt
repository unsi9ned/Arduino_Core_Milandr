# Global properties provided by the Arduino IDE
#
# {runtime.platform.path}     - the absolute path of the platform folder (i.e. the folder containing boards.txt)
# {runtime.hardware.path}     - the absolute path of the hardware folder (i.e. the folder containing the current platform folder)
# {runtime.ide.path}          - the absolute path of the Arduino IDE folder
# {runtime.ide.version}       - the version number of the Arduino IDE as a number (this uses two digits per version
#                               number component, and removes the points and leading zeroes, so Arduino IDE 1.8.3
#                               becomes 01.08.03 which becomes runtime.ide.version=10803).
# {ide_version}               - Compatibility alias for {runtime.ide.version}
# {runtime.os}                - the running OS ("linux", "windows", "macosx")
#
#
# {build.path}              - The path to the temporary folder to store build artifacts
# {build.project_name}      - The project name
# {build.arch}              - The MCU architecture (avr, sam, etc...)

################
# General Info #
################
name=Milandr MDR32FxFI
version=0.1.0

#######################################
# Extra flags that can be overwritten #
#######################################
build.extra_flags=

compiler.c.extra_flags=
compiler.cpp.extra_flags=
compiler.S.extra_flags=

##########################
# Compiler configuration #
##########################
#compiler.path={runtime.tools.xpack-arm-none-eabi-gcc.path}/bin/
compiler.path={runtime.tools.xpack-arm-none-eabi-gcc-14.2.1-1.1.path}/bin/
compiler.cpp.cmd=arm-none-eabi-g++
compiler.c.cmd=arm-none-eabi-gcc
compiler.S.cmd=arm-none-eabi-gcc
compiler.c.elf.cmd=arm-none-eabi-gcc
compiler.objcopy.cmd=arm-none-eabi-objcopy
compiler.objcopy.hex.cmd=arm-none-eabi-objcopy
compiler.size.cmd=arm-none-eabi-size
compiler.ar.cmd=arm-none-eabi-ar

# compile Include 
compiler.milandr.extra_include= "-I{build.core.path}" "-I{build.core.path}/milandr"

# compile warning
compiler.warning_flags=-w
compiler.warning_flags.none=-w
compiler.warning_flags.default=
compiler.warning_flags.more=-Wall
compiler.warning_flags.all=-Wall -Wextra

# Базовые флаги для Cortex-M3
compiler.arm.cortex.flags=-mcpu=cortex-m3 -mthumb

# Флаги C
compiler.c.flags=-c {compiler.warning_flags} -O0 -g3 -ffunction-sections -fdata-sections -std=gnu11 {compiler.arm.cortex.flags} {compiler.c.extra_flags} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_{build.arch} -D__IN_ECLIPSE__=1 {compiler.milandr.extra_include} {build.extra_flags}

# Флаги C++
compiler.cpp.flags=-c {compiler.warning_flags} -O0 -g3 -ffunction-sections -fdata-sections -fno-rtti -fno-exceptions -std=gnu++11 {compiler.arm.cortex.flags} {compiler.cpp.extra_flags} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_{build.arch} -D__IN_ECLIPSE__=1 {compiler.milandr.extra_include} {build.extra_flags}

# Флаги ассемблера
compiler.S.flags=-c -x assembler-with-cpp {compiler.arm.cortex.flags} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_{build.arch} -D__IN_ECLIPSE__=1 {compiler.milandr.extra_include}

# Флаги линкера
compiler.c.elf.flags=-O0 -g3 -ffunction-sections -fdata-sections -Wl,--gc-sections --specs=nano.specs --specs=nosys.specs {compiler.arm.cortex.flags}

# Флаги архиватора
compiler.ar.flags=rcs

#######################
# Build recipes       #
#######################

# Препроцессор C
recipe.preproc.includes="{compiler.path}{compiler.c.cmd}" {compiler.c.flags} -M -MG {includes} "{source_file}"

# Компиляция C
recipe.c.o.pattern="{compiler.path}{compiler.c.cmd}" {compiler.c.flags} -MMD -c "{source_file}" -o "{object_file}"

# Компиляция C++
recipe.cpp.o.pattern="{compiler.path}{compiler.cpp.cmd}" {compiler.cpp.flags} -MMD -c "{source_file}" -o "{object_file}"

# Компиляция ассемблера
recipe.S.o.pattern="{compiler.path}{compiler.S.cmd}" {compiler.S.flags} -MMD -c "{source_file}" -o "{object_file}"

# Линковка
recipe.c.combine.pattern="{compiler.path}{compiler.c.elf.cmd}" {compiler.c.elf.flags} -T "{build.variant.path}/ldscript.ld" -o "{build.path}/{build.project_name}.elf" {object_files} -Wl,--start-group {archive_file_path} -lm -Wl,--end-group

# Создание HEX
recipe.objcopy.hex.pattern="{compiler.path}{compiler.objcopy.hex.cmd}" -O ihex "{build.path}/{build.project_name}.elf" "{build.path}/{build.project_name}.hex"

# Подсчёт размера
recipe.size.pattern="{compiler.path}{compiler.size.cmd}" --format=berkeley "{build.path}/{build.project_name}.elf"

# Архивация
recipe.ar.pattern="{compiler.path}{compiler.ar.cmd}" {compiler.ar.flags} "{archive_file_path}" "{object_file}"

#######################
# Uploader tools      #
#######################

##CMSIS-DAP upload
tools.xpack-openocd.cmd=openocd
tools.xpack-openocd.path={runtime.tools.xpack-openocd.path}/bin
tools.xpack-openocd.upload.params.verbose=-d2
tools.xpack-openocd.upload.params.quiet=-d0
tools.xpack-openocd.upload.pattern="{path}/{cmd}" {upload.params.verbose} -f "interface/cmsis-dap.cfg" -c "transport select swd" -f "target/mdr32f9q2i.cfg" -c "program {{build.path}/{build.project_name}.hex} verify reset exit"
